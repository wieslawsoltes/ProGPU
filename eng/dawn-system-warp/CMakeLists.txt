cmake_minimum_required(VERSION 3.28)
project(ProGPUDawnSystemWarp LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The original packaged Dawn companion requires the Windows MSVC ABI")
endif()
foreach(required DAWN_SOURCE DAWN_GENERATED DAWN_IMPORT_LIBRARY)
    if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
        message(FATAL_ERROR "Missing original ${required}")
    endif()
endforeach()
add_library(progpu_dawn_system_warp SHARED progpu_dawn_system_warp.cpp)
target_compile_features(progpu_dawn_system_warp PRIVATE cxx_std_20)
set_target_properties(progpu_dawn_system_warp PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(progpu_dawn_system_warp PRIVATE /W4 /WX /permissive-)
target_compile_definitions(progpu_dawn_system_warp PRIVATE
    DAWN_NATIVE_SHARED_LIBRARY WGPU_SHARED_LIBRARY WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(progpu_dawn_system_warp SYSTEM PRIVATE
    "${DAWN_SOURCE}/include" "${DAWN_GENERATED}/include")
target_link_libraries(progpu_dawn_system_warp PRIVATE "${DAWN_IMPORT_LIBRARY}" dxgi)
